set(ORIGINAL_ENGINE "${CMAKE_CURRENT_LIST_DIR}/../../../upstream/fm1-synthdef/synthdef-engine/src")
set(PROFILE_SCRIPT "${CMAKE_CURRENT_LIST_DIR}/../../../host/engine_profile.py")
set(ENGINE "${CMAKE_CURRENT_BINARY_DIR}/target-profile/src")
file(GLOB ENGINE_INPUTS "${ORIGINAL_ENGINE}/*.c" "${ORIGINAL_ENGINE}/*.h" "${ORIGINAL_ENGINE}/*.inc")
set(PROFILE_INPUTS "${PROFILE_SCRIPT}")
foreach(PROFILE_MODULE signal_buffers dsp_kernels dsp_candidates dsp_binding dsp_routing dsp_chaos_cache dsp_core)
    list(APPEND PROFILE_INPUTS "${CMAKE_CURRENT_LIST_DIR}/../../../host/${PROFILE_MODULE}.py")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${ENGINE_INPUTS} ${PROFILE_INPUTS})
set(ENGINE_PROFILE compact)
if(CONFIG_SYNTHDEF_FM1_COMPAT)
    set(ENGINE_PROFILE fm1)
endif()
set(SYNTHDEF_DSP_CANDIDATES "none" CACHE STRING "Diagnostic candidate set: none, all, or comma-separated smooth,binary,noise,binding,routing,chaos,core")
set(PROFILE_FLAGS)
set(SYNTHDEF_BINDING_FAMILIES "" CACHE STRING "Optional comma-separated binding entry points")
if(SYNTHDEF_BINDING_FAMILIES)
    list(APPEND PROFILE_FLAGS --binding-families "${SYNTHDEF_BINDING_FAMILIES}")
endif()
set(CANDIDATE_NAMES smooth binary noise binding routing chaos core)
if(SYNTHDEF_DSP_CANDIDATES STREQUAL "all")
    set(ENABLED_CANDIDATES ${CANDIDATE_NAMES})
elseif(NOT SYNTHDEF_DSP_CANDIDATES STREQUAL "none")
    string(REPLACE "," ";" ENABLED_CANDIDATES "${SYNTHDEF_DSP_CANDIDATES}")
endif()
foreach(CANDIDATE IN LISTS ENABLED_CANDIDATES)
    if(NOT CANDIDATE IN_LIST CANDIDATE_NAMES)
        message(FATAL_ERROR "Unknown DSP candidate: ${CANDIDATE}")
    endif()
    list(APPEND PROFILE_FLAGS "--candidate-${CANDIDATE}")
endforeach()
execute_process(COMMAND "${PYTHON}" "${PROFILE_SCRIPT}" --profile "${ENGINE_PROFILE}" --output "${ENGINE}" ${PROFILE_FLAGS} RESULT_VARIABLE PROFILE_RESULT)
if(NOT PROFILE_RESULT EQUAL 0)
    message(FATAL_ERROR "Target engine capacity generation failed")
endif()
set(ENGINE_SOURCES "${ENGINE}/fm_load.c" "${ENGINE}/fm_render.c")
if(CONFIG_SYNTHDEF_DIAGNOSTICS)
    configure_file("${ENGINE}/fm_profile_render.inc" "${CMAKE_CURRENT_BINARY_DIR}/fm_profile_render.c" COPYONLY)
    list(APPEND ENGINE_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/fm_profile_render.c")
endif()
idf_component_register(SRCS ${ENGINE_SOURCES}
                       INCLUDE_DIRS "${ENGINE}" PRIV_REQUIRES esp_hw_support LDFRAGMENTS "hot_data.lf")
target_compile_options(${COMPONENT_LIB} PRIVATE -std=c99 -fno-fast-math -ffp-contract=off)
# The pinned source is tested with Clang. GCC 14 diagnoses its compact formatting,
# array spelling, and parser-validated arithmetic input counts. Keep diagnostics
# visible without making these three upstream warning classes fatal to this port.
target_compile_options(${COMPONENT_LIB} PRIVATE -Wno-error=misleading-indentation
    -Wno-error=array-parameter -Wno-error=maybe-uninitialized)
